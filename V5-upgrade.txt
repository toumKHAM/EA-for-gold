#include <Trade\Trade.mqh>
CTrade trade;

//--- Input Parameters
input group "--- EA Settings ---"
double   InpTargetProfit   = 0.2;     // ເປົ້າໝາຍກຳໄລລວມ (USD) เพื่อ Close All
double   InpGapBuy         = 1;      // ໄລຍະຫ່າງ Buy Stop (ເປັນລາຄາ ຫຼື 20 pips/points ປັບຕາມຄູ່ເງິນ)
double   InpGapSell        = 1;      // ໄລຍະຫ່າງ Sell Stop
ulong    InpMagicNumber    = 123456;   // Magic Number ຂອງ EA

//--- Global Variables
bool isFirstRun = true;

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
   trade.SetExpertMagicNumber(InpMagicNumber);
   isFirstRun = true;
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                 |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   // ບໍ່ມີຫຍັງພິເສດຕອນປິດຕົວ
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
   //--- Step 4: ກວດສອບກຳໄລລວມ > 50 USD ແລ້ວ Close All
   if(GetTotalProfit() >= InpTargetProfit)
   {
      Print("ກຳໄລຮອດເປົ້າໝາຍແລ້ວ: ", GetTotalProfit(), " USD. ກຳລັງປິດອໍເດີ້ທັງໝົດ...");
      CloseAllOrders();
      isFirstRun = true; // ຫຼັງຈາກປິດໝົດແລ້ວ ໃຫ້ເລີ່ມ Step 1 ໃໝ່
      return; 
   }

   //--- ຖ້າປິດອໍເດີ້ໝົດແລ້ວ ຫຼື ເປັນການລັນເທື່ອທຳອິດ
   if(isFirstRun && CountActivePositions() == 0)
   {
      double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      
      if(ask == 0 || bid == 0) return;

      if(IsTradeTime()){
         //--- Step 1: Buy Now 0.01 ແລະ Sell Now 0.01
         if(trade.Buy(0.01, _Symbol, ask, 0, 0, "Initial Buy") && 
            trade.Sell(0.01, _Symbol, bid, 0, 0, "Initial Sell"))
         {
            Print("ເລີ່ມຕົ້ນລະບົບ: ອອກໄມ້ Buy 0.01 ແລະ Sell 0.01 ສຳເລັດ.");
            
            //--- ຕັ້ງຄ່າລ໋ອດຕາມເງື່ອນໄຂ (ເພີ່ມເທື່ອລະ 0.03 ທຸກໆ 3 ໄມ້ ຈົນຮອດ 0.22)
            double lotPattern[] = {
               0.01, 0.01, 
               0.04, 0.04, 0.04, 
               0.08, 0.08, 0.08,  
               0.13, 0.13, 0.13,  
               0.19, 0.19, 0.19, 
               0.26, 0.26, 0.26,
               0.34, 0.34, 0.34,
               0.43, 0.43, 0.43,
               0.53, 0.53, 0.53,
               0.64, 0.64, 0.64,
               0.76, 0.76, 0.76         };
            int totalPendingOrders = ArraySize(lotPattern);
   
            //--- Step 2: ຕັ້ງ Buy Stop ຂຶ້ນໄປເລື້ອຍໆ
            double nextBuyPrice = ask;
            for(int i = 0; i < totalPendingOrders; i++)
            {
               nextBuyPrice += InpGapBuy;
               trade.BuyStop(lotPattern[i], nextBuyPrice, _Symbol, 0, 0, ORDER_TIME_GTC, 0, "Grid BuyStop");
            }
   
            //--- Step 3: ຕັ້ງ Sell Stop ລົງໄປເລື້ອຍໆ
            double nextSellPrice = bid;
            for(int i = 0; i < totalPendingOrders; i++)
            {
               nextSellPrice -= InpGapSell;
               trade.SellStop(lotPattern[i], nextSellPrice, _Symbol, 0, 0, ORDER_TIME_GTC, 0, "Grid SellStop");
            }
            
            isFirstRun = false; // ປ່ຽນສະຖານະເພື່ອບໍ່ໃຫ້ມັນອອກຊ້ຳອີກ
         }
      }
   }
}

//+------------------------------------------------------------------+
//| Function ຄິດໄລ່ກຳໄລລວມ ຂອງ Magic Number ນີ້                       |
//+------------------------------------------------------------------+
double GetTotalProfit()
{
   double total_profit = 0;
   int total_positions = PositionsTotal();
   
   for(int i = total_positions - 1; i >= 0; i--)
   {
      ulong ticket = PositionGetTicket(i);
      if(ticket > 0)
      {
         if(PositionGetString(POSITION_SYMBOL) == _Symbol && PositionGetInteger(POSITION_MAGIC) == InpMagicNumber)
         {
            total_profit += PositionGetDouble(POSITION_PROFIT) + PositionGetDouble(POSITION_SWAP);
         }
      }
   }
   return total_profit;
}

//+------------------------------------------------------------------+
//| Function ນັບຈຳນວນອໍເດີ້ທີ່ເປີດຢູ່ (Positions)                         |
//+------------------------------------------------------------------+
int CountActivePositions()
{
   int count = 0;
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == _Symbol && PositionGetInteger(POSITION_MAGIC) == InpMagicNumber)
      {
         count++;
      }
   }
   return count;
}

//+------------------------------------------------------------------+
//| Function ປິດອໍເດີ້ທັງໝົດ (ທັງ Pending ແລະ Active Positions)        |
//+------------------------------------------------------------------+
void CloseAllOrders()
{

   // 1. ປິດ Active Positions (Buy / Sell) ທັງໝົດ
   int total_positions = PositionsTotal();
   for(int i = total_positions - 1; i >= 0; i--)
   {
      ulong ticket = PositionGetTicket(i);
      if(ticket > 0)
      {
         if(PositionGetString(POSITION_SYMBOL) == _Symbol && PositionGetInteger(POSITION_MAGIC) == InpMagicNumber)
         {
            trade.PositionClose(ticket);
         }
      }
   }
   
   // 2. ລຶບ Pending Orders (Buy Stop / Sell Stop) ທັງໝົດກ່ອນ
   int total_orders = OrdersTotal();
   for(int i = total_orders - 1; i >= 0; i--)
   {
      ulong ticket = OrderGetTicket(i);
      if(ticket > 0)
      {
         if(OrderGetString(ORDER_SYMBOL) == _Symbol && OrderGetInteger(ORDER_MAGIC) == InpMagicNumber)
         {
            trade.OrderDelete(ticket);
         }
      }
   }
   
   if(PositionsTotal() > 0){
      CloseAllOrders();
   }

   
}

bool IsTradeTime()
{
   datetime now = TimeCurrent();   // server time
   //Print("------------------- ",now);
   MqlDateTime t;
   TimeToStruct(now, t);

   int minutesNow   = t.hour * 60 + t.min;
   int minutesStart = 1 * 60 + 10;    // 01:10
   int minutesEnd   = 22 * 60 + 30 ;   // 22:30

   return (minutesNow >= minutesStart && minutesNow <= minutesEnd);
}
