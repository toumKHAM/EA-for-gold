#include <Trade\Trade.mqh>
CTrade trade;


int OnInit()
{
   Print("---- EA-V100 start --- ", _Symbol);
   return(INIT_SUCCEEDED);
}

void OnDeinit(const int reason)
{

   Print("---- EA-V100 close --- ");
}

void OnTick()
{
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);

   
   //NewCandle()==true &&
   if( PositionsTotal()==0 && OrdersTotal()==0 ){
      double target_buy_stop = ask + 1;
      double lot_size = GetLotSize();
      trade.BuyStop(lot_size, target_buy_stop, _Symbol, target_buy_stop - 4, 0, NULL);
      
      double target_sell_stop = bid - 1;
      trade.SellStop(lot_size, target_sell_stop, _Symbol, target_sell_stop + 4, 0, NULL); 
      
      Print("--- Lot: ", GetTotalClosedLots());
   }
   
   if(PositionsTotal()==1){
      DeletePendingOrders();
   }
   
   if(PositionsTotal()==1){
      TrailingStepProfit();
   }
}

bool NewCandle(){
   static datetime last_time = 0;
   datetime current_time = iTime(_Symbol, _Period, 0);
   if(current_time != last_time)
   {
      last_time = current_time;
      return true;
   }
   return false;
}


double GetLotSize(){
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double raw_lot = balance * 0.00003;
   double lot_size = NormalizeDouble(raw_lot, 2);
   if(lot_size > 100){
      lot_size=100;
   }else if(lot_size < 0.01){
      lot_size=0.01;
   }
   return lot_size;
}

void DeletePendingOrders()
{
   for(int i = OrdersTotal() - 1; i >= 0; i--)
   {
      ulong ticket = OrderGetTicket(i);
      if(OrderSelect(ticket))
      {
         trade.OrderDelete(ticket);;
      }
   }
}

void TrailingStepProfit()
{
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      ulong ticket = PositionGetTicket(i);
      if(ticket)
      {
         if( PositionGetString(POSITION_SYMBOL) == _Symbol)
         {
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            double currentSL = PositionGetDouble(POSITION_SL);
            double curPrice  = PositionGetDouble(POSITION_PRICE_CURRENT);
            ENUM_POSITION_TYPE type = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
            
            double profit = 0;
            double newSL = 0;

            if(type == POSITION_TYPE_BUY)
            {
               profit = curPrice - openPrice;
               if(profit >= 1.0)
               {
                  double step = MathFloor((profit - 1.0) / 0.1) * 0.1;
                  newSL = openPrice + step;
                  if(newSL > currentSL || currentSL == 0)
                  {
                     trade.PositionModify(ticket, NormalizeDouble(newSL, _Digits), PositionGetDouble(POSITION_TP));
                  }
               }
            }
            else if(type == POSITION_TYPE_SELL)
            {
               profit = openPrice - curPrice;
               
               if(profit >= 1.0)
               {
                  double step = MathFloor((profit - 1.0) / 0.1) * 0.1;
                  newSL = openPrice - step;
                  if(newSL < currentSL || currentSL == 0)
                  {
                     trade.PositionModify(ticket, NormalizeDouble(newSL, _Digits), PositionGetDouble(POSITION_TP));
                  }
               }
            }
         }
      }
   }
}

double GetTotalClosedLots()
{
   double totalLots = 0;
   if(HistorySelect(0, TimeCurrent()))
   {
      int totalDeals = HistoryDealsTotal();
      for(int i = 0; i < totalDeals; i++)
      {
         ulong ticket = HistoryDealGetTicket(i);

         if(HistoryDealGetString(ticket, DEAL_SYMBOL) == _Symbol)
         {
            long entryType = HistoryDealGetInteger(ticket, DEAL_ENTRY);
            
            if(entryType == DEAL_ENTRY_OUT || entryType == DEAL_ENTRY_OUT_BY)
            {
               double volume = HistoryDealGetDouble(ticket, DEAL_VOLUME);
               totalLots += volume;
            }
         }
      }
   }
   
   return totalLots;
}
