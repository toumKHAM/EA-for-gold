#include <Trade\Trade.mqh>
CTrade trade;
CTrade tradeBuy;

long MagicBuy   = 222;

double my_profit = 15;
double my_gap = 10;
double lot_size = 0.01;

int OnInit()
{
   Print("EA-V1-Buy-Stop-MagicNumber: Started .... ");
   tradeBuy.SetExpertMagicNumber(MagicBuy);
   //EventSetTimer(24000);
   return(INIT_SUCCEEDED);
}
  
void OnDeinit(const int reason)
{
   Print("EA-V1-Buy-Stop-MagicNumber: Closeed .... ");
   //EventKillTimer();
}

void OnTick()
{
      string symbol = _Symbol;
      double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      
      BuyInfoV1 buy = GetBuyInfoV1();
      if(buy.min_buy_price != 99999){
         // for buy stop
         double target_stop = buy.max_buy_price + my_gap;
         if(buy.min_buy_stop == 99999){ // no buy stop
            tradeBuy.BuyStop(lot_size, target_stop, symbol, 0, target_stop+my_profit, NULL); 
         }
         
         double buy_stop_price = buy.min_buy_price - my_gap;
         if(ask < buy_stop_price){
            buy_stop_price = (buy.min_buy_price < buy.min_buy_stop) ? (buy.min_buy_price - my_gap) : buy.min_buy_stop - my_gap;
            tradeBuy.BuyStop(lot_size, buy_stop_price, symbol, 0, buy_stop_price+my_profit, NULL); 
         }
         
      }else{
         Print("No Order , Buy Now---->: ", ask);
         tradeBuy.Buy(lot_size, symbol, ask, 0, ask+my_profit);
      }
      
      buy = GetBuyInfoV1();
      ModifyMinBuyTp(buy.min_buy_price);
}

struct BuyInfoV1
{
   double max_buy_price;
   double min_buy_price;
   double min_buy_stop;
};

BuyInfoV1 GetBuyInfoV1()
{

    BuyInfoV1 info;
    info.max_buy_price  = -1;
    info.min_buy_price = 99999;
    
    
    //--- for position
    int position_total = PositionsTotal();
    for (int i = 0; i < position_total; i++){
      if (PositionGetTicket(i)){
         long type = PositionGetInteger(POSITION_TYPE);
         long magicNumber = PositionGetInteger(POSITION_MAGIC);
         if (type == POSITION_TYPE_BUY && magicNumber == MagicBuy ) 
         {
            double price = PositionGetDouble(POSITION_PRICE_OPEN);
            if (price > info.max_buy_price){
               info.max_buy_price = price;
            }
            if(price < info.min_buy_price){
               info.min_buy_price = price;
            }
         }
      }
    }
    
    info.min_buy_stop   = 99999;
    
    //--- for pending order
    int pending_total = OrdersTotal();
    for (int i = 0; i < pending_total; i++){
      if (OrderGetTicket(i)){
         int type = (int)OrderGetInteger(ORDER_TYPE);
         double price = OrderGetDouble(ORDER_PRICE_OPEN);
         long magicNumber = OrderGetInteger(ORDER_MAGIC);
         
         if (type == ORDER_TYPE_BUY_STOP && magicNumber == MagicBuy && price < info.min_buy_stop)
         {
            info.min_buy_stop = price;
         }
      }
    }
    
    return info;
}

void ModifyMinBuyTp(double minBuyPrice){
   int total = PositionsTotal();
   for(int i = 0; i < total; i++)
   {
      if(PositionGetTicket(i)) 
      {
         long   type  = PositionGetInteger(POSITION_TYPE);
         double price = PositionGetDouble(POSITION_PRICE_OPEN);
         long magicNumber = PositionGetInteger(POSITION_MAGIC);
         
         if(type == POSITION_TYPE_BUY && magicNumber==MagicBuy && price == minBuyPrice ){
            ulong  ticket = PositionGetInteger(POSITION_TICKET);
            //Print("ticket: ",ticket," , price: ",price);
            double old_tp = PositionGetDouble(POSITION_TP);
            double new_tp = price + my_profit;
            if(NormalizeDouble(new_tp, _Digits) != NormalizeDouble(old_tp, _Digits)){
               tradeBuy.PositionModify(ticket, 0, new_tp);
            }
            return;
         }
      }
    }
}
