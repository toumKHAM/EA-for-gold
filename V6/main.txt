#include <Trade\Trade.mqh>

CTrade trade;


double my_gap = 3;
double my_profit = 4;

int OnInit()
{
   return(INIT_SUCCEEDED);
}
void OnDeinit(const int reason){

}

void OnTick(){
   string symbol = _Symbol;
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   
   PositionV6 position = GetPositionV6();
   /*
   Print("-- ---- ---- ---- ----");
   Print("buy_min_price: ", position.buy_min_price);
   Print("sell_max_price: ", position.sell_max_price);
   Print("max_lot_size: ", position.max_lot_size);
   Print("has_buy_stop: ", position.has_buy_stop);
   Print("has_sell_stop: ", position.has_sell_stop);
   Print("type: ", position.type);
   */
   
   if(position.count > 0){
      if( position.type ==1 && position.has_sell_stop == false){
         trade.SellStop(2*position.max_lot_size, position.buy_min_price-my_gap, symbol, 0, 0, NULL);
      }
      if( position.type ==2 && position.has_buy_stop == false){
         trade.BuyStop(2*position.max_lot_size, position.sell_max_price+my_gap, symbol, 0, 0, NULL);
      }
      
      if(position.type ==1){ // buy
         if(bid - position.buy_min_price > my_profit){
            CloseAll(symbol);
         }
      }
      if(position.type == 2){ // sell
         if(position.sell_max_price - ask > my_profit){
            CloseAll(symbol);
         }
      }
   }else{
      if(position.has_buy_stop || position.has_sell_stop){
         DeleteStop();
      }
      
      Print("No Order , Buy Now---->: ", ask);
      trade.Buy(0.01, symbol, ask, 0, 0);
   }
   
}

struct PositionV6
{
   int count;
   double buy_min_price;
   double sell_max_price;
   double max_lot_size;
   bool has_buy_stop;
   bool has_sell_stop;
   int type;
};


PositionV6 GetPositionV6()
{
    PositionV6 info;
    info.count = 0;
    info.buy_min_price = 99999;
    info.sell_max_price = -1;
    info.max_lot_size = 0;
    info.type = 1;
    
    
    int total = PositionsTotal();

    for(int i = 0; i < total; i++)
    {
        if(PositionGetTicket(i)) 
        {
            long   type       = PositionGetInteger(POSITION_TYPE);
            double volume = PositionGetDouble(POSITION_VOLUME);
            double price = PositionGetDouble(POSITION_PRICE_OPEN);

            info.count++; 
            if (type == POSITION_TYPE_SELL && price > info.sell_max_price){
               info.sell_max_price = price;
            }
            if(type == POSITION_TYPE_BUY && price < info.buy_min_price){
               info.buy_min_price = price;
            }
            if (volume > info.max_lot_size){
               info.max_lot_size = volume;
               if(type == POSITION_TYPE_BUY){ info.type = 1; }
               else if(type == POSITION_TYPE_SELL){ info.type = 2; }
            }
            
        }
    }
    
    info.has_buy_stop = false;
    info.has_sell_stop = false;
    
    int total2 = OrdersTotal();
    for (int i = 0; i < total2; i++)
    {
      if (OrderGetTicket(i))
      {
         if ((int)OrderGetInteger(ORDER_TYPE) == ORDER_TYPE_SELL_STOP)
         {
            info.has_sell_stop = true;
         }
         if ((int)OrderGetInteger(ORDER_TYPE) == ORDER_TYPE_BUY_STOP)
         {
            info.has_buy_stop = true;
         }
      }
    }
    
    return info;
}

void CloseAll(string symbol)
{
   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; i--)
   {
      if(PositionGetTicket(i))
      {
         trade.PositionClose(symbol);
      }
   }
}

void DeleteStop()
{
   int total = OrdersTotal();
   for(int i = total - 1; i >= 0; i--)
   {
      ulong ticket = OrderGetTicket(i);
      if(OrderSelect(ticket))
      {
         trade.OrderDelete(ticket);
         Print("DeleteStop");
      }
   }
}
